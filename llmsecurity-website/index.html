Simple Integration

Add security to your AI app in minutes

from llm_firewall import LLMFirewall, FirewallConfig

# Initialize firewall
config = FirewallConfig(
    enable_prompt_injection_detection=True,
    enable_pii_redaction=True,
    enable_audit_logging=True,
)

firewall = LLMFirewall(config)

# Protect your LLM calls
async with firewall.protect(user_id="user123") as session:
    response = await session.chat(
        provider="openai",
        model="gpt-4",
        messages=[{"role": "user", "content": user_input}]
    )

# Firewall automatically:
# ✓ Blocks prompt injection
# ✓ Redacts PII
# ✓ Logs interactions
# ✓ Enforces rate limits
